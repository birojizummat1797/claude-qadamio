const NL = String.fromCharCode(10); // yangi qator (backslash ishlatmaslik uchun)
const EASE = ['Oson va qiziq', 'Oson, lekin zerikarli', 'Qiyin, lekin qiziq', 'Qiyin va zerikarli'];

function addChoices_(form, items, start) {
  items.forEach(function (it, i) {
    form.addMultipleChoiceItem()
      .setTitle((start + i) + '. ' + it.q)
      .setChoiceValues(it.o)
      .setRequired(true);
  });
}

function createQadamPilotForm() {
  const form = FormApp.create('Qadam — o‘zingizni kashf qiling (sinov)');
  form.setDescription(
    'Assalomu alaykum! Qadam — insonga o‘zini yaxshiroq tushunish va o‘ziga mos yo‘nalishni topishda ' +
    'yordam beradigan loyiha. Yangi savollarimizni sinab ko‘ryapmiz.' + NL + NL +
    '• 15–20 daqiqa.' + NL + '• Ismingiz so‘ralmaydi; javoblar faqat savollarni yaxshilash uchun ishlatiladi va hech kimga berilmaydi.' + NL +
    '• Sinov faqat 18 yoshdan kattalar uchun.' + NL + '• To‘g‘ri yoki noto‘g‘ri javob yo‘q — o‘zingizga eng yaqinini tanlang.');
  form.setCollectEmail(false);
  form.setAllowResponseEdits(false);
  form.setLimitOneResponsePerUser(false); // true bo'lsa Google akkaunt talab qilinadi — anonimlik uchun false
  form.setProgressBar(true);
  form.setShuffleQuestions(false);
  form.setConfirmationMessage('Rahmat! 🙏 Natijangizni o‘zingiz yozgan kod so‘z orqali yuboramiz.');

  // 0. Rozilik va yosh (shartli o'tish)
  const consent = form.addMultipleChoiceItem()
    .setTitle('Sinovda qatnashishga rozimisiz va 18 yoshdan kattamisiz?')
    .setRequired(true);
  const p1 = form.addPageBreakItem().setTitle('1-qism: 13 savol');
  form.addTextItem()
    .setTitle('O‘zingiz o‘ylab topgan kod so‘z (masalan: lola27). Natijangizni shu kod bilan olasiz. Ism yozmang.')
    .setRequired(true);
  addChoices_(form, DATA.short, 1);
  addChoices_(form, DATA.ctx, DATA.short.length + 1);

  form.addPageBreakItem().setTitle('2-qism: sayt va ilovalar haqida (11 savol)');
  addChoices_(form, DATA.dev, 1);

  form.addPageBreakItem().setTitle('3-qism: ish uslubi va imkoniyatlaringiz (8 savol)');
  addChoices_(form, DATA.style, 1);

  form.addPageBreakItem().setTitle('4-qism: 4 ta kichik topshiriq')
    .setHelpText('Shoshilmang, kalkulyatorsiz, o‘zingiz yeching. Bilmasangiz — «Bilmayman» ham to‘g‘ri javob.');
  DATA.tasks.forEach(function (t, i) {
    form.addMultipleChoiceItem().setTitle((i + 1) + '. ' + t.q).setChoiceValues(t.o).setRequired(true);
    form.addMultipleChoiceItem().setTitle((i + 1) + '-topshiriq siz uchun qanday bo‘ldi?').setChoiceValues(EASE).setRequired(true);
  });

  form.addPageBreakItem().setTitle('5-qism: yangi narsa o‘rganamiz')
    .setHelpText('📘 Qisqa dars: Robotga yangi buyruq o‘rgatamiz — TAKRORLA.' + NL +
                 'Masalan: TAKRORLA 3 marta: [1 qadam yur] — robot 3 qadam yuradi.');
  form.addMultipleChoiceItem()
    .setTitle('1. TAKRORLA 2 marta: [2 qadam oldinga, 1 qadam orqaga]. Robot boshlang‘ich joyidan necha qadam oldinda bo‘ladi?')
    .setChoiceValues(['2', '3', '4', '6', 'Bilmayman']).setRequired(true);
  form.addMultipleChoiceItem().setTitle('1-topshiriq siz uchun qanday bo‘ldi?').setChoiceValues(EASE).setRequired(true);
  form.addMultipleChoiceItem()
    .setTitle('2. TAKRORLA 3 marta: [ TAKRORLA 2 marta: [1 qadam yur] ]. Robot jami necha qadam yuradi?')
    .setChoiceValues(['3', '5', '6', '9', 'Bilmayman']).setRequired(true);
  form.addMultipleChoiceItem().setTitle('2-topshiriq siz uchun qanday bo‘ldi?').setChoiceValues(EASE).setRequired(true);
  form.addMultipleChoiceItem().setTitle('Darsni qayta o‘qishga to‘g‘ri keldimi?').setChoiceValues(['Ha', 'Yo‘q']).setRequired(true);

  form.addPageBreakItem().setTitle('Oxirgi savollar (qisqa, o‘z so‘zingiz bilan)');
  [
    'Qaysi savolni tushunish qiyin bo‘ldi? (qism va raqamini yozing)',
    'Qaysi savolda javoblar bir-biriga juda o‘xshab qoldi?',
    'Biror savolda «to‘g‘ri javob qaysi ekan?» deb o‘yladingizmi? Qaysida?',
    'Topshiriqlar qanday bo‘ldi: oson / me’yorida / qiyin / qo‘rqinchli?',
    'Qisqa dars (TAKRORLA) tushunarli bo‘ldimi?',
    'O‘zingiz uchun eng to‘g‘ri yo‘nalish qaysi deb o‘ylaysiz va nega?',
    'Hozir qaysi yo‘nalishda o‘qiyapsiz va undan qanchalik mamnunsiz? (juda / o‘rtacha / unchalik emas)'
  ].forEach(function (q) { form.addParagraphTextItem().setTitle(q).setRequired(false); });

  // 18 dan kichiklar uchun yakuniy sahifa
  const stop = form.addPageBreakItem().setTitle('Rahmat!')
    .setHelpText('Bu sinov faqat 18 yoshdan kattalar uchun. Qiziqishingiz uchun katta rahmat!');
  // Oddiy ketma-ketlikda bu sahifaga yetganlar (ya'ni testni tugatganlar) shu yerda formani yuboradi;
  // 18 ga to'lmaganlar esa birinchi sahifadan to'g'ridan-to'g'ri shu sahifaga o'tadi.
  stop.setGoToPage(FormApp.PageNavigationType.SUBMIT);

  consent.setChoices([
    consent.createChoice('Ha, roziman va 18 yoshdan kattaman', p1),
    consent.createChoice('Yo‘q / 18 yoshga to‘lmaganman', stop)
  ]);

  Logger.log('Forma (tahrirlash): ' + form.getEditUrl());
  Logger.log('Forma (qatnashchilar uchun havola): ' + form.getPublishedUrl());
}
